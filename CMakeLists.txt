cmake_minimum_required(VERSION 3.28)
project(Helix.RHI LANGUAGES CXX)

if(NOT COMMAND helix_add_library)
    include("${CMAKE_CURRENT_SOURCE_DIR}/External/Helix.CMake/Helix.cmake")
endif()

helix_require(Core)
helix_require(Platform)
helix_require(GL)
helix_require(D3D11)
helix_require(Profiler)
helix_add_library(helix_rhi ALIAS RHI PUBLIC_DEPS Helix::Core Helix::Platform Helix::GL PRIVATE_DEPS Helix::D3D11 Helix::Profiler)
if(HELIX_PLATFORM STREQUAL "windows")
    target_link_libraries(helix_rhi PRIVATE user32 gdi32)
endif()

helix_add_test_suite(helix_rhi_tests DEPS Helix::RHI Helix::Profiler EXCLUDE "/Support/DiscreteGpu.cpp$")
if(TARGET helix_rhi_tests AND HELIX_PLATFORM STREQUAL "windows")
    get_target_property(rhi_test_sources helix_rhi_tests SOURCES)
    helix_add_executable(helix_rhi_tests_discrete_gpu SOURCES ${rhi_test_sources} Tests/Support/DiscreteGpu.cpp DEPS Helix::RHI Helix::Profiler Helix::TestMain)
    target_include_directories(helix_rhi_tests_discrete_gpu PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/Source ${CMAKE_CURRENT_SOURCE_DIR}/Tests)
    add_test(NAME helix_rhi_tests_discrete_gpu COMMAND helix_rhi_tests_discrete_gpu)
endif()

if(PROJECT_IS_TOP_LEVEL)
    helix_add_executable(helix_rhi_triangle SOURCES Examples/Triangle/Triangle.cpp DEPS Helix::RHI)
    if(HELIX_PLATFORM STREQUAL "web")
        set_target_properties(helix_rhi_triangle PROPERTIES SUFFIX ".html")
        target_link_options(helix_rhi_triangle PRIVATE -sENVIRONMENT=web --shell-file "${CMAKE_CURRENT_SOURCE_DIR}/Examples/Shared/WebShell.html")
    endif()
endif()
